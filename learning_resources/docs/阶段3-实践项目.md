## 3. 实现自定义算子

### 3.1 在native_functions.yaml中定义

**位置**: `aten/src/ATen/native/native_functions.yaml`

```yaml
- func: my_add(Tensor self, Tensor other, Scalar alpha=1) -> Tensor
  variants: function
  dispatch:
    CPU: my_add_cpu
    CUDA: my_add_cuda
  tags: pointwise
```

### 3.2 实现CPU版本

**创建文件**: `aten/src/ATen/native/MyOps.cpp` (示例)

```cpp
#include <ATen/ATen.h>
#include <ATen/TensorIterator.h>
#include <ATen/native/cpu/Loops.h>

namespace at {
namespace native {

Tensor my_add_cpu(const Tensor& self, const Tensor& other, const Scalar& alpha) {
  Tensor result = at::empty_like(self);

  // 使用TensorIterator处理broadcast、dtype等
  auto iter = TensorIteratorConfig()
    .add_output(result)
    .add_input(self)
    .add_input(other)
    .build();

  // AT_DISPATCH处理不同数据类型
  AT_DISPATCH_ALL_TYPES_AND2(kHalf, kBFloat16, iter.dtype(), "my_add_cpu", [&] {
    scalar_t alpha_val = alpha.to<scalar_t>();
    cpu_kernel(iter, [=](scalar_t a, scalar_t b) -> scalar_t {
      return a + alpha_val * b;
    });
  });

  return result;
}

} // namespace native
} // namespace at
```

> **重要**: 直接用 `data_ptr<float>()` 只适用于float类型的连续张量。生产代码必须使用 `TensorIterator` + `AT_DISPATCH_*` 处理dtype/layout/broadcast。

### 3.3 实现CUDA版本

**创建文件**: `aten/src/ATen/native/cuda/MyOps.cu` (示例)

```cpp
#include <ATen/ATen.h>
#include <ATen/cuda/CUDAContext.h>

namespace at {
namespace native {

template<typename scalar_t>
__global__ void my_add_kernel(
    scalar_t* result,
    const scalar_t* self_data,
    const scalar_t* other_data,
    scalar_t alpha,
    int64_t n) {
  int idx = blockIdx.x * blockDim.x + threadIdx.x;
  if (idx < n) {
    result[idx] = self_data[idx] + alpha * other_data[idx];
  }
}

Tensor my_add_cuda(const Tensor& self, const Tensor& other, const Scalar& alpha) {
  Tensor result = at::empty_like(self);
  int64_t n = self.numel();

  const int threads = 256;
  const int blocks = (n + threads - 1) / threads;

  AT_DISPATCH_FLOATING_TYPES(self.scalar_type(), "my_add_cuda", [&] {
    my_add_kernel<scalar_t><<<blocks, threads>>>(
      result.data_ptr<scalar_t>(),
      self.data_ptr<scalar_t>(),
      other.data_ptr<scalar_t>(),
      alpha.to<scalar_t>(),
      n
    );
  });

  return result;
}

} // namespace native
} // namespace at
```

## 4. 为算子添加Autograd支持

### 4.1 方法1: derivatives.yaml (推荐)

**位置**: `tools/autograd/derivatives.yaml`

```yaml
- name: my_add(Tensor self, Tensor other, Scalar alpha=1) -> Tensor
  self: handle_r_to_c(self.scalar_type(), grad)
  other: handle_r_to_c(other.scalar_type(), maybe_multiply(grad, alpha.conj()))
```

> **注意**:
> - `handle_r_to_c` 处理实数到复数的转换
> - `alpha.conj()` 处理复数alpha的共轭
> - 简单的 `grad * 1` 在复数dtype上会出错

这会自动生成反向传播代码。

### 4.2 方法2: 手动实现

**示例代码** (通常通过derivatives.yaml自动生成，这里仅展示结构):

```cpp
struct MyAddBackward : public Node {
  MyAddBackward(Scalar alpha) : alpha_(alpha) {}

  variable_list apply(variable_list&& grads) override {
    auto grad_output = grads[0];

    variable_list result(2);
    result[0] = grad_output;              // grad_self = grad_output
    result[1] = grad_output * alpha_;     // grad_other = grad_output * alpha

    return result;
  }

  Scalar alpha_;
};
```

## 5. 使用 torch.library API

### 5.1 定义算子

```python
import torch
from torch import Tensor

# 定义schema
torch.library.define(
    "mylib::my_add",
    "(Tensor self, Tensor other, float alpha=1.0) -> Tensor"
)

# 实现CPU版本
@torch.library.impl("mylib::my_add", "CPU")
def my_add_cpu(self: Tensor, other: Tensor, alpha: float = 1.0) -> Tensor:
    return self + alpha * other

# 实现CUDA版本
@torch.library.impl("mylib::my_add", "CUDA")
def my_add_cuda(self: Tensor, other: Tensor, alpha: float = 1.0) -> Tensor:
    return self + alpha * other  # 可以调用C++实现

# 定义autograd
@torch.library.impl_abstract("mylib::my_add")
def my_add_abstract(self: Tensor, other: Tensor, alpha: float = 1.0) -> Tensor:
    return torch.empty_like(self)
```

### 5.2 使用自定义算子

```python
# 调用

y = torch.randn(3, 3)

```

## 7. 测试和验证

### 7.1 编写单元测试

**创建文件**: `test/test_my_ops.py`

```python
import unittest
import torch

class TestMyOps(unittest.TestCase):
    def test_my_add_cpu(self):
        x = torch.randn(3, 3)
        y = torch.randn(3, 3)
        alpha = 2.0

        result = torch.my_add(x, y, alpha)
        expected = x + alpha * y

        self.assertTrue(torch.allclose(result, expected))

    def test_my_add_cuda(self):
        if not torch.cuda.is_available():
            self.skipTest("CUDA not available")

        x = torch.randn(3, 3, device='cuda')
        y = torch.randn(3, 3, device='cuda')
        alpha = 2.0

        result = torch.my_add(x, y, alpha)
        expected = x + alpha * y

        self.assertTrue(torch.allclose(result, expected))

    def test_my_add_backward(self):
        x = torch.randn(3, 3, requires_grad=True)
        y = torch.randn(3, 3, requires_grad=True)
        alpha = 2.0

        z = torch.my_add(x, y, alpha)
        loss = z.sum()
        loss.backward()

        # 验证梯度
        self.assertIsNotNone(x.grad)
        self.assertIsNotNone(y.grad)
        self.assertTrue(torch.allclose(x.grad, torch.ones_like(x)))
        self.assertTrue(torch.allclose(y.grad, torch.ones_like(y) * alpha))

if __name__ == '__main__':
    unittest.main()
```

**运行测试**:
```bash
python test/test_my_ops.py
```

## 8. 性能优化

### 8.1 Benchmark

```python
import time
import torch

def benchmark(func, *args, warmup=10, iterations=100):
    # Warmup
    for _ in range(warmup):
        func(*args)

    torch.cuda.synchronize() if torch.cuda.is_available() else None

    start = time.time()
    for _ in range(iterations):
        func(*args)
    torch.cuda.synchronize() if torch.cuda.is_available() else None
    end = time.time()

    avg_time = (end - start) / iterations * 1000  # ms
    return avg_time

# 测试
x = torch.randn(1000, 1000, device='cuda')
y = torch.randn(1000, 1000, device='cuda')

time_native = benchmark(torch.add, x, y, 2.0)
time_custom = benchmark(torch.my_add, x, y, 2.0)

print(f"Native add: {time_native:.3f} ms")
print(f"Custom add: {time_custom:.3f} ms")
```
