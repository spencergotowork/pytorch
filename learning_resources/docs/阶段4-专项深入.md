### 3. DDP (DistributedDataParallel)

**文件**: `torch/nn/parallel/distributed.py`

**核心机制**:

```python
class DistributedDataParallel(Module):
    def __init__(self, module, device_ids, ...):
        self.module = module
        self.process_group = ...

        # 注册梯度钩子
        for param in module.parameters():
            param.register_hook(self._make_hook(param))

    def _make_hook(self, param):
        def hook(grad):
            # 累积梯度到bucket
            self.buckets[param].append(grad)

            # 如果bucket满了,进行all-reduce
            if self.buckets[param].is_ready():
                self._all_reduce_bucket(self.buckets[param])

        return hook

    def _all_reduce_bucket(self, bucket):
        # 合并bucket中的所有梯度
        flat_grad = torch.cat([g.flatten() for g in bucket.grads])

        # All-reduce
        dist.all_reduce(flat_grad, op=dist.ReduceOp.SUM)

        # 除以world_size得到平均梯度
        flat_grad /= self.world_size

        # 分发回各参数
        ...
```

**Bucketing策略**:
- 将多个小梯度合并成大bucket
- 减少通信次数
- 提高带宽利用率

### 4. FSDP (FullyShardedDataParallel)

**文件**: `torch/distributed/fsdp/fully_sharded_data_parallel.py`

**核心思想**: 将模型参数分片到各进程

```python
class FullyShardedDataParallel(Module):
    def __init__(self, module, ...):
        # 分片参数
        self._shard_parameters()

    def _shard_parameters(self):
        for param in self.module.parameters():
            # 计算分片
            numel_per_rank = param.numel() // self.world_size
            start = self.rank * numel_per_rank
            end = start + numel_per_rank

            # 保存分片
            self.sharded_param = param.data[start:end].clone()

            # 释放原参数
            param.data = torch.empty(0)

    def forward(self, *args):
        # All-gather参数
        self._all_gather_parameters()

        # 前向计算
        output = self.module(*args)

        # 释放gathered参数
        self._free_full_parameters()

        return output

    def _all_gather_parameters(self):
        for param_name, sharded_param in self.sharded_params.items():
            # All-gather收集完整参数
            full_param = torch.empty(original_shape)
            dist.all_gather_into_tensor(full_param, sharded_param)

            # 恢复到模块
            self.module.get_parameter(param_name).data = full_param
```

## 方向B: 编译器

### 1. torch.compile 管线

`torch.compile` 是PyTorch 2.0的核心功能，将前面学习的dispatcher和autograd知识与现代编译栈连接起来。

**编译管线**:
```
torch.compile(model)
    ↓
Dynamo (字节码捕获)
  → 将Python代码转为FX Graph
    ↓
AOTAutograd (前向+反向分离)
  → 生成前向和反向的独立图
    ↓
PrimTorch (算子降级)
  → 将高级算子分解为基础算子
    ↓
Inductor (代码生成)
  → 生成优化的C++/Triton代码
```

void ConstantPropagation(std::shared_ptr<Graph>& graph) {
  for (Node* node : graph->nodes()) {
    // 如果所有输入都是常量
    if (allInputsAreConstants(node)) {
      // 直接计算结果
      auto result = evaluateNode(node);

      // 替换node为常量
      node->output()->replaceAllUsesWith(result);

      // 删除node
      node->destroy();
    }
  }
}

**死代码消除** (`dead_code_elimination.cpp`):
```cpp
void EliminateDeadCode(const std::shared_ptr<Graph>& graph) {
  // 从输出开始标记live节点
  std::unordered_set<Node*> live_nodes;
  std::queue<Node*> queue;

  for (Value* output : graph->outputs()) {
    queue.push(output->node());
  }

  while (!queue.empty()) {
    Node* node = queue.front();
    queue.pop();

    if (live_nodes.insert(node).second) {
      for (Value* input : node->inputs()) {
        queue.push(input->node());
      }
    }
  }

  // 删除非live节点
  for (Node* node : graph->nodes()) {
    if (live_nodes.count(node) == 0) {
      node->destroy();
    }
  }
}
```

class InstructionTranslator:
    def step(self):
        inst = self.instructions[self.instruction_pointer]

        if inst.opname == "CALL_FUNCTION":
            return self.CALL_FUNCTION(inst)
        elif inst.opname == "BINARY_ADD":
            return self.BINARY_ADD(inst)

    def CALL_FUNCTION(self, inst):
        args = self.popn(inst.argval)
        fn = self.pop()

        # 如果是torch函数,记录到图中
        if is_torch_function(fn):
            result = self.graph.create_node("call_function", fn, args)
            self.push(result)
        else:
            # 否则graph break
            self.graph_break()

### 3. 优化技巧

**共享内存**:
```cuda
__global__ void reduce_sum(float* output, const float* input, int n) {
    __shared__ float sdata[256];

    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + tid;

    // 加载到共享内存
    sdata[tid] = (idx < n) ? input[idx] : 0.0f;
    __syncthreads();

    // 归约
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) {
            sdata[tid] += sdata[tid + s];
        }
        __syncthreads();
    }

    if (tid == 0) {
        output[blockIdx.x] = sdata[0];
    }
}
```

### 4. Triton (Python CUDA)

**文件**: `torch/_inductor/codegen/triton.py`

```python
import triton
import triton.language as tl

@triton.jit
def add_kernel(
    x_ptr, y_ptr, output_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    x = tl.load(x_ptr + offsets, mask=mask)
    y = tl.load(y_ptr + offsets, mask=mask)
    output = x + y

    tl.store(output_ptr + offsets, output, mask=mask)
```
